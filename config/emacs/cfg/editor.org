* Better defaults

#+begin_src emacs-lisp
  ;;; Create for backup and autosave
  ;; Under the XDG state directory so they stay off the home directory.
  (require 'xdg)
  (let ((state (expand-file-name "emacs" (xdg-state-home))))
    (make-directory (expand-file-name "backups" state) t)
    (make-directory (expand-file-name "autosaves" state) t))

  (setq-default

   ;; Disable splash screen
   inhibit-splash-screen t

   ;; Don't display comp warnings
   warning-suppress-log-types '((comp))
   ;; Don't create lockfiles
   create-lockfiles nil

   ;; prefer UTF8
   buffer-file-coding-system 'utf-8-unix
   default-file-name-coding-system 'utf-8-unix
   default-keyboard-coding-system 'utf-8-unix
   default-process-coding-system '(utf-8-unix . utf-8-unix)
   default-sendmail-coding-system 'utf-8-unix
   default-terminal-coding-system 'utf-8-unix

   ;; Add newline at bottom of file
   require-final-newline t

   ;; Set backup directory
   backup-directory-alist
   `(("." . ,(expand-file-name "emacs/backups" (xdg-state-home))))
   auto-save-file-name-transforms
   `((".*" ,(expand-file-name "emacs/autosaves/" (xdg-state-home)) t))

   backup-by-copying t
   delete-old-versions t
   kept-new-versions 6
   kept-old-versions 2
   version-control t

   tramp-backup-directory-alist backup-directory-alist
   tramp-auto-save-directory (expand-file-name "emacs/autosaves/"
                                               (xdg-state-home))

   ;; Skip prompt for xref find definition
   xref-prompt-for-identifier nil

   ;; Don't wait for keystrokes display
   echo-keystrokes 0.01

   ;; Disable margin for overline and underline
   overline-margin 0
   underline-minimum-offset 0

   ;; Allow resizing frame by pixels
   frame-resize-pixelwise t

   ;; Better scroll behavior
   mouse-wheel-scroll-amount '(1 ((shift) . 1) ((control) . nil))
   mouse-wheel-progressive-speed nil

   ;; Scroll line by line instead of recentring the whole window when the
   ;; cursor reaches the edge.  101 means "never recentre".
   scroll-conservatively 101
   scroll-margin 3

   ;; Disable copy region blink
   copy-region-blink-delay 0

   ;; Use short answer when asking yes or no
   read-answer-short t

   ;; Always follow link
   vc-follow-symlinks t

   ;; Disable ring bell
   ring-bell-function 'ignore

   ;; Mouse yank at current point
   mouse-yank-at-point t

   ;; DWIM target for dired
   ;; Automatically use another dired buffer as target for copy/rename
   dired-dwim-target t

   ;; Use spaces instead of tab
   indent-tabs-mode nil

   ;; Don't echo multiline eldoc
   eldoc-echo-area-use-multiline-p nil

   ;; No cursor blink in TUI
   visible-cursor nil

   ;; Counter for isearch
   isearch-lazy-count t
   isearch-lazy-highlight t

   compilation-scroll-output 'first-error

   ;; Flatten imenu for easy searching
   imenu-flatten t

   ;; For scrolling performance
   fast-but-imprecise-scrolling t
   inhibit-compacting-font-caches t
   auto-window-vscroll nil

   ;; preserve scrolling position
   scroll-preserve-screen-position t

   ;; Wider fill column
   fill-column 100

   show-paren-style 'mixed

   mode-line-compact t

   ;; Skip fontification while keyboard input is pending, so typing is not
   ;; blocked by redisplay work.
   redisplay-skip-fontification-on-input t

   ;; Cheaper bidi scanning.  The supported alternative to setting
   ;; `bidi-display-reordering' to nil, which the Emacs manual forbids.
   bidi-inhibit-bpa t

   ;; Borrowed from Doom: skip the second case-insensitive pass over
   ;; `auto-mode-alist', and skip rendering cursors/regions in unfocused
   ;; windows.
   auto-mode-case-fold nil
   cursor-in-non-selected-windows nil
   highlight-nonselected-windows nil

   ;; Read more from subprocesses per chunk (default 4kb).
   read-process-output-max (* 64 1024)

   ;; Accept y/n instead of spelling out "yes"/"no".
   use-short-answers t

   ;; Highlight the matching paren sooner (default 0.1).
   show-paren-delay 0.05)

  (when window-system
    (scroll-bar-mode -1)
    ;; Pixel-granular scrolling for trackpads and precise mice.
    (pixel-scroll-precision-mode 1))

  (global-subword-mode 1)
  (diminish 'subword-mode)

  (blink-cursor-mode -1)

  (savehist-mode)
  (recentf-mode 1)
  (global-auto-revert-mode 1)

  (add-hook 'prog-mode-hook
            (lambda ()
              (add-hook 'before-save-hook #'delete-trailing-whitespace nil t)))

  (setq jit-lock-defer-time 0)
#+end_src

* Garbage collection

`gcmh' raises the threshold while Emacs is in use and reclaims when it goes
idle, so GC pauses land between keystrokes rather than during them.  The `auto'
idle delay adapts to how busy the session is.

#+begin_src emacs-lisp
  (use-package gcmh
    :demand t
    :init
    (setq gcmh-idle-delay 'auto          ; default is a fixed 15s
          gcmh-auto-idle-delay-factor 10
          gcmh-high-cons-threshold (* 64 1024 1024))  ; 64 MB
    ;; On at startup, not during init: early-init already raised the threshold,
    ;; and letting gcmh lower it mid-init would resume collecting.
    :hook (emacs-startup . gcmh-mode)
    :config
    (diminish 'gcmh-mode))
#+end_src

* Project management

** project.el

To find files/buffers and apply commands on project, use builtin package ~project~.

#+begin_src emacs-lisp
  (require 'project)

  (setq project-switch-commands '((project-find-file "Find file")
                                  (project-find-regexp "Find regexp")
                                  (project-dired "Dired")
                                  (project-eshell "Eshell")
                                  (eat-project-other-window "Eat")
                                  (magit-project-status "Magit")
                                  (shell "Shell")))

  (keymap-set mode-specific-map "p" project-prefix-map)

  (with-eval-after-load "project"
    (keymap-set project-prefix-map "d" #'project-dired)
    (keymap-set project-prefix-map "s" #'eat-project-other-window)
    (keymap-set project-prefix-map "m" #'magit-project-status))
